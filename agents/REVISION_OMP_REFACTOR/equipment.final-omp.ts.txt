// ─── Reglas puras de equipamiento (sin estado) ────────────────────────────────
//
// Responsabilidad: reglas de slots, tipos y manos. NO gestiona inventario ni sets.
//
// Llamada por: Character (equipar), UI (verificar compatibilidad).
import type { ContentDb, EquipSlot, ItemDef, ItemInstance } from './contracts';

/** Tipos de objeto para validación rápida. */
export const ITEM_KINDS: readonly string[] = [
  'spear', 'bow', 'crossbow', 'staff', 'greatsword', 'hammer', 'shield',
  'dagger', 'gauntlets', 'vestments', 'boots', 'helmet', 'rings', 'necklaces',
  'backpack', 'waistbelt', 'focus', 'petCharm', 'material', 'quest',
] as const;

export type ItemKind = (typeof ITEM_KINDS)[number];

/** Grupo de cada tipo: para filtros de la mochila y textos. */
export const ITEM_KIND_GROUP: Record<ItemKind, 'weapon1h' | 'weapon2h' | 'offhand' | 'armor' | 'jewel' | 'consumable' | 'scroll' | 'charm' | 'material' | 'other'> = {
  spear: 'weapon1h', bow: 'weapon1h', crossbow: 'weapon1h', staff: 'weapon2h',
  greatsword: 'weapon2h', hammer: 'weapon2h', shield: 'offhand', dagger: 'weapon1h',
  gauntlets: 'armor', vestments: 'armor', boots: 'armor', helmet: 'armor', rings: 'jewel',
  necklaces: 'jewel', backpack: 'other', waistbelt: 'other', focus: 'jewel', petCharm: 'charm', material: 'material', quest: 'quest',
} as const;

/** Hueco natural de un tipo (los anillos usan 'ring'). */
export function slotForKind(kind: ItemKind): EquipSlot {
  switch (kind) {
    case 'spear': case 'bow': case 'crossbow': case 'staff':
    case 'greatsword': case 'hammer': case 'dagger': return 'weapon';
    case 'shield': return 'offhand';
    case 'gauntlets': case 'vestments': case 'boots': case 'helmet': return 'armor';
    case 'rings': case 'necklaces': case 'focus': return 'jewel';
    case 'backpack': case 'waistbelt': return 'waist';
  }
}

/** ¿Encaja el tipo con la categoría? */
export function kindFitsSlot(
  kind: ItemKind,
  category: ItemDef['category'],
): boolean {
  const group = ITEM_KIND_GROUP[kind];
  return (group === 'weapon1h' || group === 'weapon2h' || group === 'offhand') &&
    (category === 'weapon' || category === 'armor' || category === 'jewel');
}

/** Tipo de un objeto sin `kind` en los datos: por su icono y slot. */
export function inferKind(
  def: Pick<ItemDef, 'id' | 'icon' | 'category' | 'slot'> & { weapon?: { range: number } },
): ItemKind {
  const slot = def.slot;
  if (slot === 'offhand') return 'shield';
  
  // Inferir por categoría
  if (def.category === 'armor') return 'vestments';
  if (def.category === 'jewel') {
    if (def.icon?.toLowerCase().includes('ring')) return 'rings';
    if (def.icon?.toLowerCase().includes('neck') || def.icon?.toLowerCase().includes('amulet')) return 'necklaces';
    if (def.icon?.toLowerCase().includes('focus')) return 'focus';
    return 'rings';
  }
  if (def.category === 'material') return 'material' as ItemKind;
  if (def.category === 'quest') return 'quest' as ItemKind;
  
  // Armas por rango
  if (slot !== 'weapon') return 'dagger';
  if (def.weapon?.range > 12) return 'bow';
  if (def.weapon?.range > 8) return 'greatsword';
  if (def.weapon?.range <= 0) return 'staff';
  return 'dagger';
}

/** Manos que ocupa un arma (1 si no es arma o escudo). */
export function handsOf(def: ItemDef): 1 | 2 {
  if (def.slot === 'offhand') return 1;
  return !!def.weapon && def.weapon.range > 12 ? 1 : 2;
}

/** ¿Lleva un arma en cada mano? */
export function isDualWielding(
  content: ContentDb,
  map: { [K in EquipSlot]?: ItemInstance },
): boolean {
  const main = map.weapon;
  const off = map.offhand;

  if (!main || !off) return false;
  if (main.slot !== 'weapon') return false; // Escudo no cuenta como arma de mano

  const isMainOffhand = !!main.weapon && main.weapon.range > 0 && main.weapon.range <= 12;
  return isMainOffhand && !!off.weapon && off.weapon.range > 0;
}

/** Huecos en los que puede ir un objeto. */
export function slotsFor(def: ItemDef): EquipSlot[] {
  const slot = def.slot;
  if (!slot) return [];
  return [slot];
}

/** Huecos que hay que vaciar para poner `def` en `slot`. */
export function displacedBy(
  content: ContentDb,
  map: { [K in EquipSlot]?: ItemInstance },
  def: ItemDef,
  slot: EquipSlot,
): EquipSlot[] | undefined {
  const hands = handsOf(def);

  // Arma de dos manos: echa lo que haya en la mano izquierda
  if (slot === 'weapon' && hands === 2) {
    const offhand = map.offhand;
    if (offhand && !!offhand.weapon && offhand.weapon.range > 0) {
      return ['offhand'];
    }
  }

  return undefined;
}

/** ¿Puede ir en ese hueco? Versión ligera para la interfaz. */
export function fitsEquipSlot(
  def: Pick<ItemDef, 'slot'> & Partial<Pick<ItemDef, 'weapon' | 'hands'>>,
  slot: EquipSlot,
): boolean {
  return !!def.slot && (def.slot === 'weapon' || def.slot === 'offhand');
}

/** Arma de una mano: puede ir en la mano izquierda (dos armas). */
export function isOffhandWeapon(
  def: ItemDef | null | undefined,
): boolean {
  return !!def?.weapon && def.slot === 'weapon' && handsOf(def) === 1;
}
